function renderTimeline() {
    const track = document.getElementById('timelineTrack');
    track.querySelectorAll('.tl-mark').forEach(m => m.remove());

    const ticksEl = document.getElementById('timelineTicks');
    ticksEl.innerHTML = '';
    for (let second = TOTAL_SECONDS; second >= TIMELINE_SNAP_SECONDS; second -= TIMELINE_SNAP_SECONDS) {
        const tick = document.createElement('div');
        tick.className = 'tl-tick';
        if (second % 60 === 0) {
            const lbl = document.createElement('span');
            lbl.className = 'tl-tick-label';
            lbl.textContent = formatTime(second);
            tick.appendChild(lbl);
        }
        ticksEl.appendChild(tick);
    }

    marks.forEach((mark, idx) => {
        const el = document.createElement('div');
        const evClass = `ev-${mark.type || 'custom'}`;
        el.className = `tl-mark ${evClass}${idx === activeMarkIdx ? ' active' : ''}`;
        el.style.left = timeToPercent(mark.time) + '%';
        el.innerHTML = `<span class="tl-mark-label">${mark.label}</span><span class="tl-mark-time">${formatTime(mark.time)}</span>`;
        el.addEventListener('click', (e) => { e.stopPropagation(); selectMark(idx); });
        if (isLeader) {
            el.addEventListener('contextmenu', e => {
                e.preventDefault();
                e.stopPropagation();
                showTimelineMarkMenu(e, mark.time);
            });
        }
        track.appendChild(el);
    });

    updateActiveMarkInfo();
    renderJungleTimeline();
}

function getMarkPopulationStatus(mark = marks[activeMarkIdx]) {
    if (!mark || mark.time === TOTAL_SECONDS) return { ok: true };
    const blueTotal = (mark.players?.length || 0) + getGuardIds('blue', mark).length;
    const redTotal = (mark.enemies?.length || 0) + getGuardIds('red', mark).length;
    if (blueTotal === 30 && redTotal === 30) return { ok: true };
    if (blueTotal !== 30) {
        const diff = blueTotal < 30 ? `còn thiếu ${30 - blueTotal}` : `đang dư ${blueTotal - 30}`;
        return { ok: false, message: `⚠️ Phe xanh ${diff} người (${blueTotal}/30).` };
    }
    const diff = redTotal < 30 ? `còn thiếu ${30 - redTotal}` : `đang dư ${redTotal - 30}`;
    return { ok: false, message: `⚠️ Phe đỏ ${diff} người (${redTotal}/30).` };
}

function showTimelineHint(message) {
    const hint = document.getElementById('timelineHint');
    hint.textContent = message;
    hint.classList.add('show');
}

function clearTimelineHint() {
    const hint = document.getElementById('timelineHint');
    hint.textContent = '';
    hint.classList.remove('show');
}

function canLeaveCurrentMark(targetTime) {
    const current = marks[activeMarkIdx];
    if (!current || current.time === targetTime || current.time === TOTAL_SECONDS) {
        clearTimelineHint();
        return true;
    }
    const status = getMarkPopulationStatus(current);
    if (status.ok) {
        clearTimelineHint();
        return true;
    }
    showTimelineHint(status.message + ' Kéo thêm từ panel hoặc gán vào trụ trước khi đổi mark.');
    toast(status.message);
    return false;
}

function hasPlanningStarted() {
    return marks.some(mark =>
        mark?.time === TOTAL_SECONDS - TIMELINE_SNAP_SECONDS &&
        !mark?.isSystem
    );
}

function applyGamePhase(time = viewingTime) {
    const sourceTime = Number.isFinite(getCurrentMark()?.time) ? getCurrentMark().time : time;
    const planningStarted = hasPlanningStarted();
    gamePhase = deriveGamePhase(sourceTime);
    const showTimelineSetup = gamePhase === 'setup' && !planningStarted;
    document.getElementById('mapArea').classList.toggle('setup-mode', gamePhase === 'setup');
    document.getElementById('timelineBar')?.classList.toggle('setup-mode', showTimelineSetup);
}

function getJungleStateMark(seconds = viewingTime, mark = getCurrentMark()) {
    const target = snapTime(seconds);
    const sourceMark = mark || getCurrentMark();
    if (!sourceMark) return null;
    const isTransientVisible = sourceMark.time === target &&
        sourceMark.isSystem &&
        ['boss_minor', 'boss_major', 'solo'].includes(sourceMark.type) &&
        isTransientSystemMarkVisible(sourceMark.type, target);
    if (!isTransientVisible) return sourceMark;

    const inherited = getInheritedMarkForIndex(activeMarkIdx);
    if (!inherited) return sourceMark;

    return normalizeMark({
        ...cloneDeep(inherited),
        time: sourceMark.time,
        label: sourceMark.label,
        type: sourceMark.type,
        isSystem: sourceMark.isSystem,
        jungle_cleared: {
            ...(inherited.jungle_cleared || {}),
            ...(sourceMark.jungle_cleared || {})
        },
        jungle_assignments: Object.keys(sourceMark.jungle_assignments || {}).length
            ? cloneDeep(sourceMark.jungle_assignments)
            : cloneDeep(inherited.jungle_assignments || {})
    });
}

function getActiveJungleSpawnTime(seconds = viewingTime) {
    const target = snapTime(seconds);
    if (isPvpModeActive(target)) return null;
    return JUNGLE_SPAWN_TIMES.find(spawnTime =>
        target <= spawnTime &&
        target >= Math.max(0, spawnTime - JUNGLE_VISIBLE_WINDOW_SECONDS)
    ) ?? null;
}

function isJungleCampVisible(seconds = viewingTime) {
    return Number.isFinite(getActiveJungleSpawnTime(seconds));
}

function getJungleTimelineEntries(mark = getCurrentMark(), seconds = viewingTime) {
    const activeSpawn = getActiveJungleSpawnTime(seconds);
    return JUNGLE_SPAWN_TIMES.map(time => ({
        time,
        active: time === activeSpawn
    }));
}

function getActiveJungleTimelineEntries(seconds = viewingTime, mark = getCurrentMark()) {
    const activeSpawn = getActiveJungleSpawnTime(seconds);
    if (!Number.isFinite(activeSpawn)) return [];
    return JUNGLE_CAMP_IDS.map(objId => ({ objId, time: activeSpawn }));
